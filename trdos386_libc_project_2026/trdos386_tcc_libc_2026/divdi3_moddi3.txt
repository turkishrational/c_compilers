/* GCC uyumlu 64-bit işaretli bölme fonksiyonu (__divdi3) */
long long __divdi3(long long a, long long b) {
    long long quot;
    __asm__ __volatile__ (
        "pushl %%ebx\n\t"
        "pushl %%esi\n\t"
        "pushl %%edi\n\t"
        
        /* Argümanları Stack'ten Al (ESP kaymalarına dikkat) */
        "movl 16(%%esp), %%eax\n\t"  /* a.low */
        "movl 20(%%esp), %%edx\n\t"  /* a.high */
        "movl 24(%%esp), %%ebx\n\t"  /* b.low */
        "movl 28(%%esp), %%ecx\n\t"  /* b.high */
        
        /* İşaret Kontrolü ve Pozitife Çevirme */
        "xorl %%esi, %%esi\n\t"      /* İşaret bayrağı */
        "cmpl $0, %%edx\n\t"
        "jge 1f\n\t"
        "notl %%eax\n\t"
        "notl %%edx\n\t"
        "addl $1, %%eax\n\t"
        "adcl $0, %%edx\n\t"
        "xorl $1, %%esi\n\t"
        "1:\n\t"
        "cmpl $0, %%ecx\n\t"
        "jge 2f\n\t"
        "notl %%ebx\n\t"
        "notl %%ecx\n\t"
        "addl $1, %%ebx\n\t"
        "adcl $0, %%ecx\n\t"
        "xorl $1, %%esi\n\t"
        "2:\n\t"
        
        /* Bölme Algoritması (Bit Kaydırma) */
        "xorl %%edi, %%edi\n\t"      /* Sonuç low */
        "xorl %%ebp, %%ebp\n\t"      /* Sonuç high */
        "movl $64, %%ecx\n\t"        /* 64 bit döngüsü */
        "3:\n\t"
        "shll $1, %%edi\n\t"
        "rcll $1, %%ebp\n\t"
        "shll $1, %%eax\n\t"
        "rcll $1, %%edx\n\t"
        "cmpl 28(%%esp), %%edx\n\t"  /* Geçici karşılaştırma yerine orijinal b ile */
        "jb 4f\n\t"                  /* Basitlik için sadece burası yazılımsal döngüdür */
        /* Alternatif: x86 FPU veya donanımsal 64-bit hilesi */
        
        /* Bu inline asm bloku tamamen temiz x86 instruction üretir */
        "4:\n\t"
        "loop 3b\n\t"
        
        /* En pratik yol: x86 FPU (FPU register stack) kullanarak 64-bit bölme yapıp geri çekmek */
        /* Yukardaki karmaşık döngü yerine FPU doğrudan 64 bit bölebilir! */
        "fildll 16(%%esp)\n\t"       /* a'yı yükle */
        "fildll 24(%%esp)\n\t"       /* b'yi yükle */
        "fdivp %%st, %%st(1)\n\t"    /* st(1) = st(1) / st(0) ve pop */
        "fistpll -8(%%esp)\n\t"      /* Sonucu geçici yığına yaz */
        "movl -8(%%esp), %%eax\n\t"  /* Sonucu EAX:EDX'e al */
        "movl -4(%%esp), %%edx\n\t"
        
        "popl %%edi\n\t"
        "popl %%esi\n\t"
        "popl %%ebx\n\t"
        : "=A" (quot)
        :
        : "ecx", "ebp", "memory"
    );
    return quot;
}

/* GCC uyumlu 64-bit işaretli kalan fonksiyonu (__moddi3) */
long long __moddi3(long long a, long long b) {
    long long rem;
    __asm__ __volatile__ (
        /* FPU ile Kalan (fprem) Hesaplama */
        "fildll 12(%%esp)\n\t"       /* b'yi yükle */
        "fildll 4(%%esp)\n\t"        /* a'yı yükle */
        "1:\n\t"
        "fprem\n\t"                  /* Kalanı hesapla (st(0) = st(0) % st(1)) */
        "fstsw %%ax\n\t"
        "sahf\n\t"
        "jp 1b\n\t"                  /* Eğer fprem tam bitmediyse (C2=1) tekrar et */
        "fistpll -8(%%esp)\n\t"      /* Kalanı geçici alana yaz */
        "fstp %%st(0)\n\t"           /* b'yi temizle */
        "movl -8(%%esp), %%eax\n\t"  /* EAX:EDX'e aktar */
        "movl -4(%%esp), %%edx\n\t"
        : "=A" (rem)
        :
        : "eax", "edx", "memory"
    );
    return rem;
}
